  [SerializeField] private QVModel.Data right;              // 0x7C
  [SerializeField] public float giveUpScale;                // 0xB4
  private Weapon bindWpn;                                   // 0xB8
  [SerializeField] private Transform pickUpArea;            // 0xC0

QVModel.Data 结构体：
  public Transform Model;           // 0x0
  public Vector3 Pos_OnHand;        // 0x4
  public Vector3 Euler_OnHand;      // 0x10
  public Vector3 Pos_GiveUp;        // 0x1C
  public Vector3 Euler_GiveUp;      // 0x28
  public ParticleSystem GunFire;    // 0x34

注意：QVModel 可能对应武器模型（左手/右手持枪模型），
left 和 right 分别对应左右手持枪时的模型数据。
