## 一、文档分析总结

### 1.1 UnityCrossFire.dll 的实现方式

根据文档分析，UnityCrossFire.dll 是一个已实现的外挂DLL，它通过以下方式实现武器赋予：

```
UnityCrossFire.dll (外挂)
    ↓ (通过IL2CPP反射API)
调用游戏的 GameManager.GiveWeapon 方法
    ↓
实现武器赋予功能
```

**关键发现**：
- UnityCrossFire.dll 基址: `0x10000000`
- 它内部存储了游戏函数的字符串名称（用于IL2CPP反射）
- 它通过 `il2cpp_class_get_method_from_name` 等API获取游戏函数

### 1.2 我们需要的实现方式

**不使用 IL2CPP 反射API**，而是：
```
Frida 注入脚本
    ↓ (通过NativeFunction)
直接调用 GameAssembly.dll 中的 GameManager.GiveWeapon
    ↓
实现武器赋予功能
```

## 四、实现步骤详解

### 步骤1: 获取 GameAssembly.dll 基址

**目标**: 定位游戏主模块

**实现代码**:
```javascript
var gameAssembly = null;

function getGameAssembly() {
    if (gameAssembly) return gameAssembly;
    
    var mod = Process.findModuleByName('GameAssembly.dll');
    if (!mod) {
        console.log('[ERROR] 未找到 GameAssembly.dll');
        return null;
    }
    
    gameAssembly = mod;
    console.log('[SUCCESS] GameAssembly.dll 基址: ' + mod.base);
    return mod;
}
```

**依据**: 这是所有后续操作的基础

---

### 步骤2: 获取 GameManager 静态实例

**目标**: 获取 GameManager 类的静态字段区域

**实现代码**:
```javascript
var OFF = {
    Klass_staticFields: 0x5C,  // IL2CPP结构偏移
    GM_myPlayer: 0x00,         // myPlayer字段偏移
};

function getGameManagerStaticFields() {
    var mod = getGameAssembly();
    if (!mod) return null;
    
    // 1. 读取 GameManager_TypeInfo 指针
    // RVA: 0x0E2933C (已验证)
    var typeInfoSlot = mod.base.add(0x0E2933C);
    var klass = typeInfoSlot.readPointer();
    
    if (!klass || klass.isNull()) {
        console.log('[ERROR] 无法获取 GameManager klass');
        return null;
    }
    
    // 2. 读取静态字段区域指针
    var staticFields = klass.add(OFF.Klass_staticFields).readPointer();
    
    if (!staticFields || staticFields.isNull()) {
        console.log('[ERROR] 无法获取静态字段区域');
        return null;
    }
    
    console.log('[SUCCESS] GameManager 静态字段: ' + staticFields);
    return staticFields;
}
```

**依据**: 
- 文档中提到 GameManager 是单例类
- IL2CPP 中静态字段通过 `klass + 0x5C` 访问

### 步骤4: 初始化 GiveWeapon 函数

**目标**: 创建 NativeFunction 对象

**实现代码**:
```javascript
var giveWeaponFunc = null;

function initGiveWeapon() {
    if (giveWeaponFunc) return true;
    
    var mod = getGameAssembly();
    if (!mod) return false;
    
    try {
        // 创建 NativeFunction
        // RVA: 0xAFB390 (已验证)
        // 签名: Weapon* GiveWeapon(Player*, int, bool, bool, MethodInfo*)
        giveWeaponFunc = new NativeFunction(
            mod.base.add(0xAFB390),
            'pointer',  // 返回 Weapon*
            ['pointer', 'int', 'int', 'int', 'pointer']
        );
        
        console.log('[SUCCESS] GiveWeapon 函数初始化成功');
        return true;
    } catch(e) {
        console.log('[ERROR] 初始化失败: ' + e.message);
        return false;
    }
}
```

**依据**: 
- 文档步骤4: "调用GiveWeapon方法"
- script.json 中的函数签名

### 步骤5: 调用 GiveWeapon 赋予武器

**目标**: 调用游戏函数，赋予玩家武器

**实现代码**:
```javascript
function giveWeapon(weaponId, autoGiveUp, autoSelect) {
    console.log('[INFO] === 开始赋予武器 ===');
    console.log('[INFO] 武器ID: ' + weaponId);
    
    // 1. 初始化函数
    if (!initGiveWeapon()) {
        console.log('[ERROR] 函数初始化失败');
        return false;
    }
    
    // 2. 获取玩家对象
    var player = getMyPlayer();
    if (!player) {
        console.log('[ERROR] 无法获取玩家对象');
        return false;
    }
    
    // 3. 参数类型转换
    var wpnId = parseInt(weaponId) || 0;
    var giveUp = (autoGiveUp === true || autoGiveUp === 1) ? 1 : 0;
    var select = (autoSelect === true || autoSelect === 1) ? 1 : 0;
    
    console.log('[INFO] 参数: player=' + player + ', wpnId=' + wpnId + 
                ', giveUp=' + giveUp + ', select=' + select);
    
    // 4. 调用 GiveWeapon
    try {
        var weapon = giveWeaponFunc(
            player,     // Player* 玩家对象
            wpnId,      // int 武器ID
            giveUp,     // int 是否自动放弃
            select,     // int 是否自动选择
            ptr(0)      // MethodInfo* (可选)
        );
        
        console.log('[INFO] 返回值: ' + weapon);
        
        // 5. 检查结果
        if (weapon && !weapon.isNull()) {
            console.log('[SUCCESS] ✅ 成功赋予武器!');
            return true;
        } else {
            console.log('[ERROR] ❌ 赋予武器失败');
            return false;
        }
    } catch(e) {
        console.log('[ERROR] 异常: ' + e.message);
        return false;
    }
}
```

**依据**: 
- 文档步骤4-5: "调用GiveWeapon方法" 和 "处理返回结果"
- 文档中的参数说明

## 六、完整代码示例

### 6.1 JavaScript 完整代码

```javascript
(function() {
    'use strict';

    // ==================== RVA地址常量（已验证）====================
    var RVA = {
        GameManager_TypeInfo: 0x0E2933C,
        GiveWeapon: 0xAFB390,
        GetWpnName: 0xAFB1F0,
    };

    // ==================== 字段偏移常量（已验证）====================
    var OFF = {
        Klass_staticFields: 0x5C,
        GM_myPlayer: 0x00,
    };

    // ==================== 全局变量 ====================
    var gameAssembly = null;
    var giveWeaponFunc = null;

    // ==================== 核心函数 ====================
    function getGameAssembly() {
        if (gameAssembly) return gameAssembly;
        var mod = Process.findModuleByName('GameAssembly.dll');
        if (!mod) return null;
        gameAssembly = mod;
        return mod;
    }

    function getMyPlayer() {
        var mod = getGameAssembly();
        if (!mod) return null;
        
        var typeInfoSlot = mod.base.add(RVA.GameManager_TypeInfo);
        var klass = typeInfoSlot.readPointer();
        if (!klass || klass.isNull()) return null;
        
        var staticFields = klass.add(OFF.Klass_staticFields).readPointer();
        if (!staticFields || staticFields.isNull()) return null;
        
        var myPlayer = staticFields.add(OFF.GM_myPlayer).readPointer();
        return myPlayer;
    }

    function giveWeapon(weaponId, autoGiveUp, autoSelect) {
        var mod = getGameAssembly();
        if (!mod) return false;
        
        if (!giveWeaponFunc) {
            giveWeaponFunc = new NativeFunction(
                mod.base.add(RVA.GiveWeapon),
                'pointer',
                ['pointer', 'int', 'int', 'int', 'pointer']
            );
        }
        
        var player = getMyPlayer();
        if (!player) return false;
        
        var wpnId = parseInt(weaponId) || 0;
        var giveUp = (autoGiveUp === true || autoGiveUp === 1) ? 1 : 0;
        var select = (autoSelect === true || autoSelect === 1) ? 1 : 0;
        
        var weapon = giveWeaponFunc(player, wpnId, giveUp, select, ptr(0));
        
        return weapon && !weapon.isNull();
    }

    // ==================== RPC导出 ====================
    rpc.exports = {
        giveweapon: giveWeapon,
        getmyplayer: getMyPlayer,
    };

    console.log('[SUCCESS] 武器赋予脚本已加载');
})();
```
