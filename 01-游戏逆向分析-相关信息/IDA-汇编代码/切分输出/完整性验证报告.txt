```python
import glob

def restore_file(part_name, output_file):
    chunks = sorted(glob.glob(f"切分输出/{part_name}/*.lst"))
    with open(output_file, 'w', encoding='utf-8') as out:
        for chunk in chunks:
            with open(chunk, 'r', encoding='utf-8') as in_file:
                out.write(in_file.read())

# 还原所有文件
restore_file('part1', '还原_GameAssembly_part1.lst')
restore_file('part2', '还原_GameAssembly_part2.lst')
restore_file('part3', '还原_GameAssembly_part3.lst')
```
